select dbsname, sum(ti_nptotal) as alloc,
                sum(ti_npused) as used,
                sum(ti_npdata) as data
  from systabnames, systabinfo
where partnum = ti_partnum
   and dbsname not like '%sys%'
   and dbsname not like '%dbs%'
   and dbsname not like '%HASH%'
group by 1
order by 2 desc
